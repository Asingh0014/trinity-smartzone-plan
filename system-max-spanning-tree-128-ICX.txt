!=============================================================================
! 200V switches - raise spanning-tree limit to 128 (system-max spanning-tree 128)
! Ruckus ICX, FastIron 10.0.10g
!=============================================================================
!
! Why
!   ICX switches run spanning-tree on at most 32 VLANs by default. Most 200V
!   switches now have more VLANs than that, so the extra VLANs run without
!   spanning-tree. On lv6 that includes 2600 (CCTV) and 2700 (AV).
!   128 gives room for all current VLANs plus growth.
!
! IMPORTANT: the new limit only takes effect after a reload
!   system-max values are read at boot. lv4 already has 'system-max
!   spanning-tree 40' saved but still runs spanning-tree on exactly 32 VLANs,
!   which shows the reload has not happened there. Each switch below needs a
!   reload in a maintenance window. Reloading a stack restarts every unit.
!
! Order
!   Access switches first, one at a time, smallest first. The core goes last,
!   in its own window: reloading it cuts every 200V switch off until it is back.
!   LT2-2 and LT2-3 connect through LT2-1, so they also drop while LT2-1 reloads.
!
! The ICX7150-C12 switches (LT2-2, LT2-3, switch-200V-8, Tutorial-6.03):
!   before the change, run 'system-max spanning-tree ?' in config mode and
!   confirm 128 is within the allowed range.
!
! Switch                       Mgmt IP          VLANs  STP on  Current limit
!   switch-200V-LT2-2          192.168.122.13      32      32  32 (default)
!   switch-200V-LT2-3          192.168.122.14      32      32  32 (default)
!   switch-200V-8              192.168.122.16      28      28  32 (default)
!   switch-200V-Tutorial-6.03  192.168.122.15      26      26  32 (default)
!   switch-200V-LT2-1          192.168.122.12      34      32  32 (default)
!   switch-200V-stack-lv3      192.168.122.22      35      32  32 (default)
!   switch-200V-stack-lv4      192.168.122.21      35      32  40 configured, not active until reload
!   switch-200V-stack-lv6      192.168.122.11      35      32  32 (default)
!   trinity200V-core           10.20.8.2           34      32  32 (default)
!=============================================================================

!=============================================================================
! switch-200V-LT2-2  (192.168.122.13, ICX7150-C12)
!=============================================================================
!
!-----------------------------------------------------------------------------
! PRE-CHECK
!-----------------------------------------------------------------------------
show default values
! Note the 'spanning-tree' line: current value and the value used at next boot
show running-config | include system-max
show span | include VLAN
!
!-----------------------------------------------------------------------------
! CHANGE
!-----------------------------------------------------------------------------
configure terminal
system-max spanning-tree ?
! Continue only if 128 is in the range shown
system-max spanning-tree 128
end
write memory
show default values
! The next-boot value should now read 128
!
!-----------------------------------------------------------------------------
! RELOAD (maintenance window)
!-----------------------------------------------------------------------------
reload
! Answer y to confirm. Wait for the switch to come back
!
!-----------------------------------------------------------------------------
! VERIFY (after reload)
!-----------------------------------------------------------------------------
show default values
! Current spanning-tree value should be 128
show running-config | include system-max
show interfaces brief
show lldp neighbors
!
!-----------------------------------------------------------------------------
! ROLLBACK
!-----------------------------------------------------------------------------
configure terminal
no system-max spanning-tree 128
! returns to the default of 32
end
write memory
! A reload is needed for the old limit to apply again
