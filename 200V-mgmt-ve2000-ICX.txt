!=============================================================================
! 200V access switches - move management from ve 100 to ve 2000 (VL-netmgmt-01)
! Ruckus ICX, FastIron 10.0.10g
!=============================================================================
!
! New addressing: VLAN 2000, 10.21.0.0/24, gateway 10.21.0.1 (trinity200V-core ve 2000)
! trinity200V-core is not changed (its management stays on ve 1960, 10.20.8.2)
!
!   Switch                      Now (ve 100)      New (ve 2000)  
!   switch-200V-stack-lv6       192.168.122.11/22 10.21.0.20/24  done 26 Oct
!   switch-200V-LT2-1           192.168.122.12/22 10.21.0.21/24  
!   switch-200V-LT2-2           192.168.122.13/22 10.21.0.22/24  
!   switch-200V-LT2-3           192.168.122.14/22 10.21.0.23/24  
!   switch-200V-Tutorial-6.03   192.168.122.15/22 10.21.0.24/24  
!   switch-200V-8               192.168.122.16/22 10.21.0.25/24  
!   switch-200V-stack-lv4       192.168.122.21/22 10.21.0.26/24  
!   switch-200V-stack-lv3       192.168.122.22/22 10.21.0.27/24  
!
! What changes on each switch
!   - interface ve 2000 gets the new address
!   - default route moves from 192.168.120.1 to 10.21.0.1
!   - management source-interface ve 2000 protocol all, so every management
!     protocol the switch originates (SSH/telnet client, SNMP traps, syslog,
!     NTP, DNS, TFTP/SCP, RADIUS/TACACS) uses 10.21.0.x
!   - the SmartZone connection (manager active-list) re-forms from 10.21.0.x
!   - last step: the 192.168.122.x address is removed from ve 100, so the switch
!     has no management address left on VLAN 100. VLAN 100 itself is not
!     removed (it is still tagged on ports and the uplink)
!
! Before starting
!   - DONE: 10.21.0.20 - 10.21.0.27 are excluded from the 10.21.0.0/24 DHCP scope
!     on 10.20.16.10 / 10.20.16.11 (confirmed by Trinity 22 Oct)
!   - Work from the SmartZone CLI for each switch (telnet is disabled on the
!     switches and the core cannot SSH to them). The SmartZone session DROPS at
!     STEP 3: the tunnel re-forms from 10.21.0.x in 3 - 5 minutes, then reopen the
!     CLI. Method proven on switch-200V-stack-lv6, 26 Oct 2026
!   - Safety net: 'reload after 00:00:15' (dd:hh:mm = 15 minutes) before STEP 3.
!     If the switch does not come back, it reloads with the saved (old) config.
!     Save Part A first (write memory), and run 'reload cancel' as soon as the
!     switch is back in SmartZone
!   - Pilot on switch-200V-Tutorial-6.03 first (a single switch with few users),
!     then the others
!   - These ICX run the router image: while both ve 100 and ve 2000 have an
!     address (STEP 1 to STEP 4) the switch can route between 192.168.120.0/22 and
!     10.21.0.0/24. Keep that window short; STEP 4 ends it
!
! Per switch: PRE-CHECK > STEP 1 (add ve 2000) > STEP 2 (test from core, set
! reload timer) > STEP 3 (route + source interface; session drops) > wait for
! SmartZone > reload cancel > VERIFY > write memory > STEP 4 (remove 192.168.122.x)
!=============================================================================

!=============================================================================
! trinity200V-core PRE-CHECK (read-only)
!=============================================================================
show running-config interface ve 2000
show vlan 2000
ping 10.21.0.20
ping 10.21.0.21
ping 10.21.0.22
ping 10.21.0.23
ping 10.21.0.24
ping 10.21.0.25
ping 10.21.0.26
ping 10.21.0.27
! Expect: ve 2000 10.21.0.1/24, VLAN 2000 tagged on lag 1 to 6 and lag 10,
! and NO reply from 10.21.0.20 - .27 (addresses free)

!=============================================================================
! switch-200V-stack-lv6  (192.168.122.11 -> 10.21.0.20)
!=============================================================================
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 PRE-CHECK (on the switch, current session)
!-----------------------------------------------------------------------------
show running-config interface ve 100
show running-config interface ve 2000
show ip interface
show ip route
show running-config | include ip route|management source|ip dns|snmp-server host|logging host|manager active
show vlan 2000
show manager status
! Expect: ve 100 192.168.122.11/22 is the only IP interface, default route 192.168.120.1,
! VLAN 2000 tagged on the uplink, SmartZone connected. If another IP interface
! shows (for example a DHCP-learned address), note it for STEP 4
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 STEP 1 - add ve 2000 (no impact)
!-----------------------------------------------------------------------------
configure terminal
interface ve 2000
 port-name VE-netmgmt-01
 ip address 10.21.0.20 255.255.255.0
 exit
end
ping 10.21.0.1
! Expect replies from 10.21.0.1. If not, stop: check VLAN 2000 on the uplink
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 STEP 2 - test from the core, then set the safety timer
!-----------------------------------------------------------------------------
! On trinity200V-core:
ping 10.21.0.20
! (expect replies; 'ping 192.168.245.100 source 10.21.0.1' was tested OK on 26 Oct)
! On the switch (SmartZone CLI): write memory, then
write memory
reload after 00:00:15
show reload
! Expect a reload scheduled in about 15 minutes; if not, 'reload cancel' and stop
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 STEP 3 - default route and management source (SmartZone CLI; the session drops)
!-----------------------------------------------------------------------------
configure terminal
ip route 0.0.0.0/0 10.21.0.1
no ip route 0.0.0.0/0 192.168.120.1
no management source-interface ve 100 protocol all
management source-interface ve 2000 protocol all
end
! Wait 3 - 5 minutes until SmartZone shows the switch online on 10.21.0.20, reopen its CLI
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 VERIFY
!-----------------------------------------------------------------------------
reload cancel
show reload
show ip route
show running-config | include ip route|management source|ip dns
ping 10.21.0.1
ping 10.128.128.31
ping 192.168.245.100
show manager status
! Expect: one default route via 10.21.0.1, management source ve 2000,
! replies from 10.21.0.1, DNS and SmartZone. The SmartZone connection drops
! and re-forms from the new address; allow up to 5 minutes, repeat
! show manager status until it shows connected
write memory
show ntp associations
! Expect the NTP server to become reachable within a few minutes
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 STEP 4 - remove the old address (after SmartZone shows 10.21.0.20)
!-----------------------------------------------------------------------------
configure terminal
interface ve 100
 no ip address 192.168.122.11 255.255.252.0
 exit
end
show ip interface
! Expect: ve 2000 10.21.0.20 is the only IP interface. Remove any other address
! noted in the PRE-CHECK the same way
write memory
!
!-----------------------------------------------------------------------------
! switch-200V-stack-lv6 ROLLBACK (only if needed; SmartZone CLI or console)
!-----------------------------------------------------------------------------
configure terminal
interface ve 100
 ip address 192.168.122.11 255.255.252.0
 exit
ip route 0.0.0.0/0 192.168.120.1
no ip route 0.0.0.0/0 10.21.0.1
no management source-interface ve 2000 protocol all
management source-interface ve 100 protocol all
end
write memory
! Then, once the switch is back in SmartZone on 192.168.122.11, remove ve 2000's address:
configure terminal
interface ve 2000
 no ip address 10.21.0.20 255.255.255.0
 exit
end
write memory
